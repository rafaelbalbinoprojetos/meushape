function MacroProgress({ macro, mode, accent }) {
  const min = Number.isFinite(macro.min) ? macro.min : null;
  const max = Number.isFinite(macro.max) ? macro.max : null;
  const current = Number.isFinite(macro.current) ? macro.current : null;
  const idealHigh = Number.isFinite(macro.ideal) ? macro.ideal : max ?? min ?? current;
  const target = mode === "max" ? max : max ?? min;
  const hasData = current != null && target != null;
  const percent = hasData ? Math.min(120, Math.round((current / target) * 100)) : 0;
  const statusText =
    !hasData
      ? "Sem dados"
      : mode === "max"
        ? current > max
          ? "Acima do teto"
          : "Dentro do limite"
        : current < min
          ? "Abaixo do mínimo"
          : "Dentro do alvo";
  return (
    <article className="rounded-[24px] border border-white/50 bg-white/80 p-4 shadow-inner dark:border-slate-800 dark:bg-slate-900/70">
      <p className="text-xs uppercase tracking-[0.3em] text-[rgb(var(--text-subtle))]">{macro.label}</p>
      <div className="mt-2 space-y-2 text-sm text-[rgb(var(--text-secondary))]">
        <div className="flex items-center justify-between">
          <span>Meta mínima: {min == null ? "—" : `${min}${macro.unit}`}</span>
          <span className="text-xs">{hasData && min != null ? (current >= min ? "✅" : "❌") : "—"}</span>
        </div>
        <div className="flex items-center justify-between">
          <span>
            Meta ideal: {min == null || idealHigh == null ? "—" : `${min}-${idealHigh}${macro.unit}`}
          </span>
          <span className="text-xs">—</span>
        </div>
        <div className="flex items-center justify-between">
          <span>Estado atual: {current == null ? "—" : `${current}${macro.unit}`}</span>
          <span className="text-xs">{statusText}</span>
        </div>
      </div>
      <div className="mt-3 h-3 rounded-full bg-slate-200 dark:bg-slate-800">
        <div className={`h-full rounded-full bg-gradient-to-r ${accent}`} style={{ width: `${Math.min(percent, 110)}%` }} />
      </div>
      <p className="mt-2 text-xs text-[rgb(var(--text-secondary))]">{hasData ? `${percent}% do alvo` : "—"}</p>
    </article>
  );
}

function MacroPill({ label, value }) {
  return (
    <div className="rounded-full bg-[rgba(15,31,60,0.05)] px-3 py-1 text-center font-semibold text-[rgb(var(--text-secondary))] dark:bg-white/5 dark:text-white/80">
      {label} · {value}
    </div>
  );
}
