const toNumber = (value) => {
  if (value === null || value === undefined || value === "") return null;
  const normalized = typeof value === "string" ? value.replace(",", ".") : value;
  const num = Number(normalized);
  return Number.isFinite(num) ? num : null;
};

const toInt = (value) => {
  const num = toNumber(value);
  return num === null ? null : Math.round(num);
};

const normalizeDate = (value) => {
  if (!value) return null;
  if (value instanceof Date) return value.toISOString().slice(0, 10);
  if (typeof value === "string") {
    const trimmed = value.trim();
    if (!trimmed) return null;
    if (/^\d{4}-\d{2}-\d{2}/.test(trimmed)) return trimmed.slice(0, 10);
  }
  return null;
};

const normalizeAltura = (value) => {
  const num = toNumber(value);
  if (!num) return null;
  return num > 3 ? Number((num / 100).toFixed(2)) : num;
};

const safeJsonStringify = (value, space = 0) => {
  const seen = new WeakSet();
  const sanitize = (val, depth = 0) => {
    if (val === null || val === undefined) return null;
    const type = typeof val;
    if (type === "string" || type === "number" || type === "boolean") return val;
    if (type === "function") return "[function]";
    if (typeof Element !== "undefined" && val instanceof Element) return `[element ${val.tagName?.toLowerCase?.() || "node"}]`;
    if (depth > 5) return "[depth-limit]";
    if (type === "object") {
      if (seen.has(val)) return "[circular]";
      seen.add(val);
      if (Array.isArray(val)) return val.slice(0, 50).map((item) => sanitize(item, depth + 1));
      const output = {};
      Object.keys(val)
        .slice(0, 50)
        .forEach((key) => {
          try {
            output[key] = sanitize(val[key], depth + 1);
          } catch {
            output[key] = "[unserializable]";
          }
        });
      return output;
    }
    return String(val);
  };
  try {
    const safe = sanitize(value, 0);
    return JSON.stringify(safe, null, space);
  } catch (err) {
    console.warn("[Coach] safeJsonStringify falhou:", err);
    return "[unserializable]";
  }
};

const safeStringify = (value) => {
  if (value === null || value === undefined) return "";
  if (typeof value === "string") return value;
  if (typeof value !== "object") return String(value);
  try {
    return safeJsonStringify(value, 2);
  } catch (err) {
    console.warn("[Coach] falha ao serializar detalhes:", err);
    return "[detalhes indisponíveis]";
  }
};

              content: "Relatorio de bioimpedancia salvo na evolucao. Se precisar, posso ajustar ou complementar.",
            },
          ]);
        } catch (error) {
          console.error("[Coach] falha ao salvar evolucao:", error);
          toast.error(error?.message ?? "Nao foi possivel salvar a evolucao.");
        }
        return;
      }

      if (isMealPayload) {
        console.log("[Coach] meal payload detectado", parsedJson);
        const items = parsedJson?.itens || parsedJson?.items || [];
        const totals =
          parsedJson?.valores_totais ||
          parsedJson?.totais ||
          parsedJson?.total || {
            calorias_kcal: parsedJson?.total_calorias,
            proteinas_g: parsedJson?.total_proteina_g,
            carboidratos_g: parsedJson?.total_carboidrato_g,
            gorduras_g: parsedJson?.total_gordura_g,
            calorias: parsedJson?.total?.calorias,
            proteina: parsedJson?.total?.proteina,
            carboidrato: parsedJson?.total?.carboidrato,
            gordura: parsedJson?.total?.gordura,
          };

        const sumFromItems = (key, aliases = []) =>
          items.reduce((sum, item) => {
            const candidates = [key, ...aliases];
            for (const c of candidates) {
              const val =
                Number(item?.[c]) ||
                Number(item?.[`total_${c}`]) ||
                Number(item?.[`macros_${c}`]) ||
                Number(item?.[`macros_${c}_g`]);
              if (Number.isFinite(val)) return sum + val;
            }
            return sum;
          }, 0);

        const totalCalorias = Number(
          totals?.calorias_kcal ?? totals?.calorias ?? parsedJson?.total_calorias ?? parsedJson?.total?.calorias,
        );
        const totalProteina = Number(
          totals?.proteinas ?? totals?.proteinas_g ?? totals?.proteina_g ?? totals?.proteina ?? parsedJson?.total_proteina_g ?? parsedJson?.total?.proteina,
        );
        const totalCarbo = Number(
          totals?.carboidratos ?? totals?.carboidratos_g ?? totals?.carboidrato_g ?? totals?.carboidrato ?? parsedJson?.total_carboidrato_g ?? parsedJson?.total?.carboidrato,
        );
        const totalGord = Number(
          totals?.gorduras ?? totals?.gorduras_g ?? totals?.gordura_g ?? totals?.gordura ?? parsedJson?.total_gordura_g ?? parsedJson?.total?.gordura,
        );

        const resolvedTotals = {
          calorias: Number.isFinite(totalCalorias) ? totalCalorias : sumFromItems("calorias", ["calories", "kcal"]),
          proteina: Number.isFinite(totalProteina) ? totalProteina : sumFromItems("proteina", ["proteina_g", "protein", "proteinas", "proteinas_g"]),
          carboidrato: Number.isFinite(totalCarbo) ? totalCarbo : sumFromItems("carboidrato", ["carboidrato_g", "carbo", "carboidratos", "carboidratos_g", "carbs"]),
          gordura: Number.isFinite(totalGord) ? totalGord : sumFromItems("gordura", ["gordura_g", "gorduras", "gorduras_g", "fat"]),
        };

        const filledTotals = {
          calorias: Number.isFinite(resolvedTotals.calorias) ? resolvedTotals.calorias : 0,
          proteina: Number.isFinite(resolvedTotals.proteina) ? resolvedTotals.proteina : 0,
          carboidrato: Number.isFinite(resolvedTotals.carboidrato) ? resolvedTotals.carboidrato : 0,
          gordura: Number.isFinite(resolvedTotals.gordura) ? resolvedTotals.gordura : 0,
        };
        console.log("[Coach] meal resolved totals", filledTotals);

        const macrosLine = `Macros estimados: Proteínas ${filledTotals.proteina || "?"} g | Carboidratos ${filledTotals.carboidrato || "?"} g | Gorduras ${filledTotals.gordura || "?"} g | Calorias ${filledTotals.calorias || "?"} kcal`;
        setMessages((current) => [...current, { role: "assistant", content: macrosLine }]);

        if (user?.id) {
          try {
            // salva apenas em refeicoes_consumidas
            console.log("[Coach] salvando refeicao em refeicoes_consumidas");
            await insertConsumedMeal({
              usuarioId: user.id,
              nome:
                parsedJson?.titulo ||
                parsedJson?.nome ||
                parsedJson?.summary ||
                file.name ||
                "Refeição analisada",
              tipo: parsedJson?.tipo ?? null,
              data: new Date(),
              totalCalorias: filledTotals.calorias,
              totalProteina: filledTotals.proteina,
              totalCarboidratos: filledTotals.carboidrato,
              totalGorduras: filledTotals.gordura,
            });

            toast.success("Refeição analisada e salva.");
          } catch (persistMealError) {
            console.error("[Coach] falha ao salvar refeição:", persistMealError);
            toast.error("Macros analisados, mas não foi possível salvar a refeição.");
          }
        } else {
          toast.success("Refeição analisada. Faça login para salvar no histórico.");
        }
        return;
      }

      // Desabilitado salvar em evolução a partir deste fluxo (apenas refeições/imagens); avaliações em PDF podem ser reabilitadas se necessário.
      const shouldPersistAssessment = false;
      if (!shouldPersistAssessment) return;

      if (!user?.id) {
        toast.error("Faça login para salvar os resultados em evolução.");
        return;
      }

      try {
        console.log("[Coach] salvando avaliacao via API/saveAssessment", parsedJson);
        const persistRes = await fetch(`${API_BASE}/api/saveAssessment`, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: safeJsonStringify({
            usuarioId: user.id,
            dados: parsedJson,
            fileName: file.name,
          }),
        });
        const persistData = await persistRes.json();
        console.log("[Coach] resposta saveAssessment", persistRes.status, persistData);
        if (!persistRes.ok) {
          const reason =
            (typeof persistData?.error === "string" && persistData.error) ||
            (typeof persistData?.details === "string" && persistData.details) ||
            (persistData ? safeStringify(persistData) : "Falha ao salvar.");
          throw new Error(reason);
        }
        toast.success(`Evolução salva (id ${persistData?.evolucaoId || "?"}).`);
      } catch (persistError) {
        console.error("[Coach] erro ao salvar avaliacao:", persistError);
        const reason =
          persistError?.message ||
          persistError?.error?.message ||
          persistError?.error ||
          "Nao foi possivel salvar no historico.";
        toast.error(`Analise concluida, mas falhou ao salvar: ${reason}`);
      }
    } catch (error) {
      console.error("[Coach] erro ao analisar arquivo:", error);
      toast.error(error.message || "Nao foi possivel analisar o arquivo.");
      setMessages((current) => [...current, { role: "assistant", content: "Não consegui analisar este arquivo agora." }]);
    } finally {
      setAnalyzing(false);
      bottomRef.current?.scrollIntoView({ behavior: "smooth" });
    }
  };

  const sendMessage = async (event, overrideText = null) => {
    event?.preventDefault();
    const text = typeof overrideText === "string" ? overrideText : input;
    const trimmed = text.trim();
    if (!trimmed || loading) return;

    const isRecipeMode = chatMode === "recipe" || /^\/receita\b/i.test(trimmed);
    const userContent = isRecipeMode ? trimmed.replace(/^\/receita\b[:\s]*/i, "").trim() || trimmed : trimmed;

    const nextMessages = [...messages, { role: "user", content: userContent }];
    const sanitizedMessages = nextMessages.map(({ role, content }) => {
      if (typeof content === "string") return { role, content };
      if (content?.type === "image") {
        return { role, content: content.caption || "Imagem enviada." };
      }
      return {
        role,
        // garante que não iremos serializar elementos React/DOM que causam erro de JSON circular
        content: safeStringify(content),
      };
    });
    setMessages(sanitizedMessages);
    if (!overrideText) setInput("");
    setLoading(true);
    bottomRef.current?.scrollIntoView({ behavior: "smooth" });

    try {
      const response = await fetch(`${API_BASE}/api/chat`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: safeJsonStringify({
          systemPrompt: isRecipeMode
            ? buildRecipePrompt()
            : buildSystemPrompt(context, displayName, latestEvolution, profileData, usePersonalContext),
          messages: sanitizedMessages,
          userId: user?.id ?? null,
          model: selectModel({ isLong: trimmed.length > 800 }),
        }),
      });
      const raw = await response.text();
      let data = null;
      try {
        data = raw ? JSON.parse(raw) : null;
      } catch (err) {
        console.error("[Coach] resposta não é JSON válido:", raw?.slice(0, 400));
      }
      if (!response.ok) {
        const reason = data?.error || raw || "Falha no coach.";
        throw new Error(reason);
      }
      const reply = data?.reply ?? raw ?? "Sem resposta no momento.";
      setMessages((current) => [...current, { role: "assistant", content: reply }]);

      if (isRecipeMode) {
        const recipeJson = extractJsonFromText(reply);
        if (!recipeJson) {
          toast.error("Não consegui estruturar a receita em JSON.");
          return;
        }
        if (!user?.id) {
          toast.error("Faça login para salvar a receita.");
          return;
        }
        try {
          await salvarRefeicao({
            planoId: null,
            titulo: recipeJson.titulo || recipeJson.nome || "Receita",
            descricao: recipeJson.descricao || null,
            valores: recipeJson.valores_totais || recipeJson.total || {
              calorias: recipeJson.calorias,
              proteinas: recipeJson.proteinas,
              carboidratos: recipeJson.carboidratos,
              gorduras: recipeJson.gorduras,
            },
          });
          toast.success("Receita salva em refeições.");
        } catch (err) {
          console.error("[Coach] falha ao salvar receita:", err);
          toast.error("Não foi possível salvar a receita.");
        }
        return;
      }

      if (!isRecipeMode) {
        const mealTotals = extractMealTotalsFromText(reply);
        if (mealTotals) {
          if (!user?.id) {
            toast.error("Faça login para registrar a refeição.");
          } else {
            try {
              await insertConsumedMeal({
                usuarioId: user.id,
                nome: inferMealNameFromText(trimmed),
                data: new Date(),
                horario: new Date().toTimeString().slice(0, 8),
                tipo: null,
                totalCalorias: mealTotals.calorias,
                totalProteina: mealTotals.proteina,
                totalCarboidratos: mealTotals.carboidrato,
                totalGorduras: mealTotals.gordura,
              });
              toast.success("Refeição registrada no diário.");
            } catch (persistError) {
              console.error("[Coach] falha ao salvar refeição do chat:", persistError);
              toast.error("Macros lidos, mas não foi possível salvar a refeição.");
            }
          }
        }
      }
    } catch (error) {
      console.error("[Coach] erro ao enviar mensagem:", error);
      toast.error(error.message);
      setMessages((current) => [...current, { role: "assistant", content: "Ainda não consegui responder. Tente novamente." }]);
    } finally {
      setLoading(false);
      bottomRef.current?.scrollIntoView({ behavior: "smooth" });
    }
  };

  return (
    <div className="grid h-full min-h-0 w-full grid-cols-1 gap-4 px-3 lg:grid-cols-[360px,1fr] lg:gap-6 lg:px-6">
      <div className="lg:col-span-2">
        <UserSummary context={context} />
      </div>
      {/*
        Mobile: contexto rápido em collapse no final. Desktop: painel lateral fixo.
      */}
      <aside className="hidden space-y-6 rounded-[32px] border border-white/10 bg-white/80 p-6 shadow-xl dark:border-slate-800 dark:bg-slate-900/70 lg:block">
        <ContextPanel context={context} setContext={setContext} handlePrompt={handlePrompt} />
      </aside>

      <section className="flex flex-1 min-h-0 flex-col gap-4">
        <div className="flex h-[36vh] min-h-[36vh] max-h-[36vh] flex-col gap-4 overflow-y-auto rounded-[24px] border border-white/40 bg-white/80 p-4 dark:border-slate-800 dark:bg-slate-900/60 lg:h-[520px] lg:min-h-[520px] lg:max-h-[520px] lg:max-w-full">
          {messages.map((message, index) => (
            <MessageBubble key={`${message.role}-${index}`} role={message.role} content={message.content} />
          ))}
          {loading ? <TypingIndicator /> : null}
          <span ref={bottomRef} aria-hidden />
        </div>

        <form className="mt-6 space-y-3" onSubmit={sendMessage}>
          <div className="flex items-center gap-3">
            <div className="relative flex flex-1 items-center gap-2 rounded-3xl border border-white/60 bg-white px-3 py-2 shadow-sm ring-2 ring-transparent focus-within:border-[#32C5FF] focus-within:ring-[#32C5FF]/20 dark:border-slate-800 dark:bg-slate-900">
              <button
                type="button"
                onClick={() => setAttachOpen((prev) => !prev)}
                disabled={loading || analyzing}
                className="flex h-10 w-10 items-center justify-center rounded-2xl text-[#0f1f3c] transition hover:bg-[#f0f4ff] dark:text-white dark:hover:bg-white/10 disabled:cursor-not-allowed disabled:opacity-60"
                aria-label="Abrir anexos"
                aria-expanded={attachOpen}
              >
                <ClipIcon />
              </button>
              <textarea
                value={input}
                onChange={(event) => setInput(event.target.value)}
                placeholder="Escreva uma mensagem..."
                rows={2}
                className="min-h-[56px] max-h-[240px] w-full flex-1 resize-none bg-transparent px-1 py-2 text-sm outline-none focus:outline-none dark:text-white lg:min-h-[120px]"
              />
              <button
                type="button"
              onClick={handleSelectPhoto}
              disabled={loading || analyzing}
              className="flex h-10 w-10 items-center justify-center rounded-2xl text-[#0f1f3c] transition hover:bg-[#f0f4ff] dark:text-white dark:hover:bg-white/10 disabled:cursor-not-allowed disabled:opacity-60"
              aria-label="Abrir câmera"
            >
                <CameraIcon />
              </button>
              <ClipMenu
                open={attachOpen}
                onClose={() => setAttachOpen(false)}
                onSelectGallery={() => handleSelectFile("default")}
                onSelectCamera={() => handleSelectPhoto("default")}
                onSelectDocument={handleSelectDoc}
                onSelectAudio={handleSelectAudio}
                onSelectBio={handleSelectBioReport}
              />
            </div>

            <div className="flex items-center gap-2">
              <FloatingActionButton
                hasText={Boolean(input.trim())}
                loading={loading}
                transcribing={transcribing}
                recording={recording}
                analyzing={analyzing}
                onSend={sendMessage}
                onRecord={toggleRecording}
              />
            </div>
          </div>

          <div className="flex flex-wrap items-center justify-between gap-3 text-xs text-[rgb(var(--text-secondary))]">
            <p>Enviaremos o contexto do {displayName} para respostas mais precisas.</p>
            <div className="flex flex-wrap items-center gap-2">
              <input
                ref={fileInputRef}
                type="file"
                accept="image/*,.pdf"
                onChange={handleFileChange}
                className="hidden"
              />
              <input
                ref={photoInputRef}
                type="file"
                accept="image/*"
                capture="environment"
                onChange={handlePhotoChange}
                className="hidden"
              />
              <input
                ref={docInputRef}
                type="file"
                accept=".pdf,application/pdf"
                onChange={handleDocChange}
                className="hidden"
              />
              <input
                ref={audioInputRef}
                type="file"
                accept="audio/*"
                onChange={handleAudioChange}
                className="hidden"
              />
            </div>
          </div>
        </form>
      </section>

      <div className="lg:col-span-2">
        <div className="rounded-[24px] border border-white/10 bg-white/80 p-3 shadow-md dark:border-slate-800 dark:bg-slate-900/70">
          <label className="flex flex-wrap items-center gap-3 text-sm text-[rgb(var(--text-secondary))]">
            <span className="font-semibold text-[rgb(var(--text-primary))]">Basear fichas e dietas nos dados pessoais</span>
            <select
              value={usePersonalContext ? "personal" : "free"}
              onChange={handlePersonalContextChange}
              className="rounded-2xl border border-white/60 bg-white px-3 py-2 text-sm outline-none focus:border-[#32C5FF] dark:border-slate-800 dark:bg-slate-900 dark:text-white"
            >
              <option value="personal">Sim</option>
              <option value="free">Nao</option>
            </select>
          </label>
        </div>
      </div>

      <div className="lg:hidden">
        <div className="rounded-[24px] border border-white/10 bg-white/80 p-4 shadow-md dark:border-slate-800 dark:bg-slate-900/70">
          <button
            type="button"
            onClick={() => setShowMobileContext((prev) => !prev)}
            className="flex w-full items-center justify-between rounded-2xl bg-white/70 px-4 py-3 text-left text-sm font-semibold text-[rgb(var(--text-primary))] shadow-sm transition hover:bg-white dark:bg-slate-800 dark:text-white"
          >
            <span>Contexto rápido</span>
            <span className="text-xs text-[rgb(var(--text-secondary))]">{showMobileContext ? "ocultar" : "mostrar"}</span>
          </button>
          {showMobileContext ? (
            <div className="mt-3 space-y-4">
              <ContextPanel context={context} setContext={setContext} handlePrompt={handlePrompt} />
            </div>
          ) : null}
        </div>
      </div>
    </div>
  );
}

  return (

      <div

function MessageContent({ content }) {
  if (!content) return null;
  if (typeof content !== "string") {
    if (content?.type === "image") {
      return (
        <div className="space-y-2">
          {content.caption ? <p className="leading-relaxed">{content.caption}</p> : null}
          {content.src ? (
            <img
              src={content.src}
              alt={content.caption || "Pré-visualização da imagem"}
              className="w-full max-w-[320px] rounded-2xl border border-white/40 object-cover shadow-sm dark:border-slate-700"
            />
          ) : null}
        </div>
      );
    }
    return <p className="leading-relaxed">{safeStringify(content)}</p>;
  }
  const blocks = content.split(/\n{2,}/).filter(Boolean);

  return (
    <div className="space-y-2">
      {blocks.map((block, blockIndex) => {
        const lines = block.split("\n").filter(Boolean);
        const isOrderedList = lines.every((line) => /^\d+[\.\)]\s+/.test(line.trim()));
        const isBulletList = lines.every((line) => /^[-•]\s+/.test(line.trim()));

        if (isOrderedList) {
          return (
            <ol key={`block-${blockIndex}`} className="ml-5 list-decimal space-y-1">
              {lines.map((line, lineIndex) => (
                <li key={`ol-${blockIndex}-${lineIndex}`} className="leading-relaxed">
                  {line.replace(/^\d+[\.\)]\s+/, "")}
                </li>
              ))}
            </ol>
          );
        }

        if (isBulletList) {
          return (
            <ul key={`block-${blockIndex}`} className="ml-4 list-disc space-y-1">
              {lines.map((line, lineIndex) => (
                <li key={`ul-${blockIndex}-${lineIndex}`} className="leading-relaxed">
                  {line.replace(/^[-•]\s+/, "")}
                </li>
              ))}
            </ul>
          );
        }

        // default paragraph
        return (
          <p key={`p-${blockIndex}`} className="leading-relaxed whitespace-pre-wrap">
            {block}
          </p>
        );
      })}
    </div>
  );
}

function FloatingActionButton({ hasText, loading, recording, transcribing, analyzing, onSend, onRecord }) {
  const disabled = loading || analyzing || (transcribing && !recording);
  const isSending = hasText;
  const showMic = !isSending;
  const buttonStyle = isSending ? "bg-[#0f1f3c]" : recording ? "bg-rose-600" : "bg-[#32C5FF]";

  return (
    <button
      type={isSending ? "submit" : "button"}
      onClick={isSending ? undefined : onRecord}
      disabled={disabled}
      className={`inline-flex h-12 w-12 items-center justify-center rounded-full text-white shadow-lg shadow-slate-900/30 transition hover:opacity-90 disabled:cursor-not-allowed disabled:opacity-60 ${buttonStyle}`}
      aria-label={isSending ? "Enviar mensagem" : recording ? "Parar e transcrever" : "Gravar áudio"}
    >
      {isSending ? <SendIcon /> : <MicIcon active={recording} />}
    </button>
  );
}

function extractJsonFromText(text = "") {
  if (!text) return null;
  const match = text.match(/\{[\s\S]*\}/);
  if (!match) return null;
  try {
    return JSON.parse(match[0]);
  } catch {
    return null;
  }
}

function inferMealNameFromText(text = "") {
  if (!text) return "Refeição";
  const firstLine = text.split("\n")[0] || text;
  const beforeSeparator = firstLine.split(/[:\-–]/)[0]?.trim();
  const candidate = beforeSeparator || firstLine.trim();
  return candidate?.slice(0, 60) || "Refeição";
}

function extractMealTotalsFromText(text = "") {
  if (!text) return null;
  const normalized = text.normalize("NFC");
  const matches = [...normalized.matchAll(/(calorias?|kcal|prote[ií]nas?|prot|carboidratos?|carbo|carbs?|gorduras?|gordura|fat)\s*[:=]?\s*(\d+[.,]?\d*)/gi)];
  if (matches.length === 0) {
    return null;
  }

  const totals = {
    calorias: null,
    proteina: null,
    carboidrato: null,
    gordura: null,
  };

  matches.forEach(([, label, value]) => {
    const num = Number(value.replace(",", "."));
    const rounded = Number.isFinite(num) ? Math.round(num) : null;
    if (/calorias?|kcal/i.test(label)) totals.calorias = rounded ?? totals.calorias;
    else if (/prote/i.test(label) || /^prot/i.test(label)) totals.proteina = rounded ?? totals.proteina;
    else if (/carb/i.test(label)) totals.carboidrato = rounded ?? totals.carboidrato;
    else if (/gordur|fat/i.test(label)) totals.gordura = rounded ?? totals.gordura;
  });

  const hasAny = totals.calorias !== null || totals.proteina !== null || totals.carboidrato !== null || totals.gordura !== null;
  if (!hasAny) return null;

  return {
    calorias: totals.calorias ?? 0,
    proteina: totals.proteina ?? 0,
    carboidrato: totals.carboidrato ?? 0,
    gordura: totals.gordura ?? 0,
  };
}

function buildSystemPrompt(context, displayName, latestEvolution, profileData, usePersonalContext) {
  const evoParts = [];
  if (latestEvolution) {
    if (latestEvolution.peso) evoParts.push(`Peso: ${latestEvolution.peso} kg`);
    if (latestEvolution.altura) evoParts.push(`Altura: ${latestEvolution.altura} m`);
    if (latestEvolution.imc) evoParts.push(`IMC: ${latestEvolution.imc}`);
    if (latestEvolution.gordura_corporal) evoParts.push(`Gordura corporal: ${latestEvolution.gordura_corporal}%`);
    if (latestEvolution.massa_magra) evoParts.push(`Massa magra: ${latestEvolution.massa_magra} kg`);
    const medidas = [
      latestEvolution.peito && `Peito ${latestEvolution.peito} cm`,
      latestEvolution.braco && `Braço ${latestEvolution.braco} cm`,
      latestEvolution.cintura && `Cintura ${latestEvolution.cintura} cm`,
      latestEvolution.quadril && `Quadril ${latestEvolution.quadril} cm`,
      latestEvolution.perna && `Perna ${latestEvolution.perna} cm`,
    ].filter(Boolean);
    if (medidas.length > 0) {
      evoParts.push(`Circunferências: ${medidas.join(" | ")}`);
    }
  }
  const evoSummary = evoParts.length > 0 ? evoParts.join(" | ") : "Sem registros de evolução ainda.";

  const profileParts = [];
  if (usePersonalContext && profileData) {
    if (profileData.nome) profileParts.push(`Nome completo: ${profileData.nome}`);
    if (profileData.sexo) profileParts.push(`Sexo: ${profileData.sexo}`);
    if (profileData.idade) profileParts.push(`Idade: ${profileData.idade} anos`);
    if (profileData.data_nascimento) profileParts.push(`Nascimento: ${profileData.data_nascimento}`);
    if (profileData.peso) profileParts.push(`Peso atual: ${profileData.peso} kg`);
    if (profileData.altura) profileParts.push(`Altura: ${profileData.altura} m`);
    if (profileData.objetivo) profileParts.push(`Objetivo pessoal: ${profileData.objetivo}`);
    if (profileData.nivel) profileParts.push(`Nivel: ${profileData.nivel}`);
    if (profileData.objetivo_peso_kg) profileParts.push(`Meta de peso: ${profileData.objetivo_peso_kg} kg`);
    if (profileData.meta_calorias) profileParts.push(`Meta calorias: ${profileData.meta_calorias} kcal`);
    if (profileData.meta_proteina_g) profileParts.push(`Meta proteina: ${profileData.meta_proteina_g} g`);
    if (profileData.meta_carbo_g) profileParts.push(`Meta carbo: ${profileData.meta_carbo_g} g`);
    if (profileData.meta_gordura_g) profileParts.push(`Meta gordura: ${profileData.meta_gordura_g} g`);
    if (profileData.dias_treino_semana) profileParts.push(`Dias treino/semana: ${profileData.dias_treino_semana}`);
    if (profileData.tempo_treino_min) profileParts.push(`Tempo treino: ${profileData.tempo_treino_min} min`);
    if (profileData.restricoes_alimentares) profileParts.push(`Restricoes alimentares: ${profileData.restricoes_alimentares}`);
    if (profileData.alergias) profileParts.push(`Alergias: ${profileData.alergias}`);
    if (profileData.medicacoes) profileParts.push(`Medicacoes: ${profileData.medicacoes}`);
    if (profileData.saude_observacoes) profileParts.push(`Observacoes de saude: ${profileData.saude_observacoes}`);
  }
  const profileSummary = profileParts.length > 0 ? profileParts.join(" | ") : "Sem dados pessoais adicionais.";

  return `Você é o Coach Virtual do aplicativo MEU SHAPE. Sempre responda em português, com tom motivador e direto.
Dados do usuário:
- Nome: ${displayName}
- Objetivo principal: ${context.objective}
- Calorias alvo: ${context.calories} kcal
- Equipamentos disponíveis: ${context.equipment.join(", ") || "peso corporal"}
- Evolução recente: ${evoSummary}
${usePersonalContext ? `- Dados pessoais: ${profileSummary}` : "- Dados pessoais: ignorar (criacao livre)."}
